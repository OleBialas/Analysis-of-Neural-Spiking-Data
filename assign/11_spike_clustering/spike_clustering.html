Load PCA features and reshape to a 2D matrix

pca_features = np.load("pca_features.npy")
n_spikes, n_features, n_channels = pca_features.shape
X = pca_features.reshape(-1, n_features*n_channels, order="F")

Find the correct number of clusters

Run GaussianMixture with different number of clusters and compute BIC. To make things go faster, we only take the first 1000 spikes

n_components_range = range(2, 20+1)
bic_scores = []
for n_components in n_components_range:
    gmm = GaussianMixture(
        n_components=n_components,
        covariance_type='full',  # Using full covariance matrix
        random_state=42,
        n_init=10,
        max_iter=100
    )
    gmm.fit(X[:1000])
    bic_scores.append(gmm.bic(X))
    print(f"Components: {n_components}, BIC: {gmm.bic(X):.2f}, AIC: {gmm.aic(X):.2f}")

Predict cluster labels

Fit the model with the optimal number of clusters and predict spike labels and their probabilities

best_gmm = GaussianMixture(
    n_components=n_components,
    random_state=42,
    n_init=10
)
best_gmm.fit(X)
spike_labels = best_gmm.predict(X)
spike_probs = best_gmm.predict_proba(X)

We can add the spike labels to our data frame of spike times

df = pd.read_csv("spike_times.csv")
df["Cluster"] = spike_labels
df["Probability"] = 0.0
for i_c in np.unique(spike_labels):
    idx = np.where(spike_labels==i_c)[0]
    probs = spike_probs[idx, i_c]
    df.iloc[idx, 2] = probs
df.to_csv("spike_clusters.csv", index=False)
print(f"Number of spikes in each cluster:")
for i in range(n_components):
    print(f"Cluster {i+1}: {np.sum(spike_labels == i)} spikes")

for i in range(n_components):

    print(f"Cluster {i+1}: Mean probability = {np.mean(cluster_probs):.3f}, Min = {np.min(cluster_probs):.3f}")
