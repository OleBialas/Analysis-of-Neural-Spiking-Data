Define a function to compute a correlogramm

def correlogram(spikes1, spikes2, bin_size=0.001, window=0.05):
    is_auto = np.array_equal(spikes1, spikes2)

    bins = np.arange(-window, window + bin_size, bin_size)
    bin_centers = bins[:-1] + bin_size/2
    counts = np.zeros(len(bins) - 1)

    for i, spike in enumerate(spikes1):

        mask = np.abs(spikes2 - spike) <= window
        if is_auto: # exclude self-reference
            mask[i] = False

        if np.any(mask):
            time_diffs = spikes2[mask] - spike
            hist, _ = np.histogram(time_diffs, bins=bins)
            counts += hist
    
    counts = counts / (len(spikes1))

    return bin_centers, counts

Simulate some data to test how CCG works. Let’s assume two cases: one where two neurons fire synchronously and one where they fire with a fix delay.

t = np.linspace(0, 10, 10000)

Plot the compute and plot the CCG for both cases

Autocorrelogram

Simulate two independent neurons with a refractory period of 2s

def generate_spikes_with_refractory(duration=10, rate=10, refractory=0.002):
    spikes = []
    t = 0
    while t < duration:
        interval = np.random.exponential(1/rate)
        interval = max(interval, refractory)
        t += interval
        if t < duration:
            spikes.append(t)
    return np.array(spikes)
spikes1 = generate_spikes_with_refractory(rate=50)
spikes2 = generate_spikes_with_refractory(rate=50)
fig, ax = plt.subplots(1,2, sharex=True, sharey=True)
bins, probabilities = correlogram(spikes1, spikes2, bin_size=0.0005, window=0.03)
ax[0].bar(bins * 1e3, probabilities, width=1.0, align='center')
bins, probabilities = correlogram(spikes1, spikes1, bin_size=0.0005, window=0.03)
ax[1].bar(bins * 1e3, probabilities, width=1, align='center')
ax[0].set(xlabel="Time Lag [ms]", ylabel="Spike Coincidence [a.u.]", title="Cross-Correlogram")
ax[1].set(xlabel="Time Lag [ms]", title="Auto-Correlogram", xlim=(np.min(bins*1e3), np.max(bins*1e3)))
plt.show()
